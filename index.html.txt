        function playSfx(type) {
            if (!soundEnabled) return;
            if (audioCtx.state === 'suspended') audioCtx.resume();
            
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            
            const now = audioCtx.currentTime;
            if (type === 'click') {
                osc.type = 'square';
                osc.frequency.setValueAtTime(440, now);
                osc.frequency.exponentialRampToValueAtTime(880, now + 0.08);
                gain.gain.setValueAtTime(0.1, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);
                osc.start(now);
                osc.stop(now + 0.08);
            } else if (type === 'success') {
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(523.25, now);

        // Mini Game Logic: Flappy Ghast
        function initFlappy() {
            const canvas = document.getElementById('flappy-canvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            let y = 160, vel = 0, gravity = 0.6, jump = -8, score = 0, isOver = false;
            let pipes = [{x: 320, top: 120, bottom: 120}];

            function loop() {
                if (isOver) return;
                ctx.fillStyle = '#121214';
                ctx.fillRect(0, 0, 320, 380);

                vel += gravity;
                y += vel;

                ctx.fillStyle = '#e2e8f0';
                ctx.fillRect(50, y, 28, 28);
                ctx.fillStyle = '#ef4444';
                ctx.fillRect(58, y + 8, 4, 4);
                ctx.fillRect(66, y + 8, 4, 4);

                ctx.fillStyle = '#22c55e';
                pipes.forEach(p => {
                    p.x -= 2;
                    ctx.fillRect(p.x, 0, 44, p.top);
                    ctx.fillRect(p.x, 380 - p.bottom, 44, p.bottom);

                    if (p.x < 50 && p.x > 30 && (y < p.top || y > 380 - p.bottom)) {
                        isOver = true;
                        playSfx('explosion');
                    }
                });

                if (pipes[pipes.length - 1].x < 160) {
                    const top = Math.random() * 120 + 40;
                    pipes.push({x: 320, top: top, bottom: 120});
                }
                if (pipes[0].x < -50) pipes.shift();

                if (y > 380 || y < 0) {
                    isOver = true;
                    playSfx('explosion');
                }

                if (!isOver) requestAnimationFrame(loop);
                else {
                    ctx.fillStyle = '#ffffff';
                    ctx.font = 'bold 20px Inter';
                    ctx.fillText("Game Over!", 100, 190);
                }
            }

            canvas.onclick = () => {
                if (!isOver) {
                    vel = jump;
                    playSfx('click');
                }
            };
            loop();
        }

        // Mini Game Logic: Boat Snake
        function initSnake() {
            const canvas = document.getElementById('snake-canvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            let snake = [{x: 160, y: 160}], dx = 16, dy = 0, food = {x: 80, y: 80}, score = 0, isOver = false;

            window.onkeydown = e => {
                if (e.key === 'ArrowUp' && dy === 0) { dx = 0; dy = -16; }
                if (e.key === 'ArrowDown' && dy === 0) { dx = 0; dy = 16; }
                if (e.key === 'ArrowLeft' && dx === 0) { dx = -16; dy = 0; }
                if (e.key === 'ArrowRight' && dx === 0) { dx = 16; dy = 0; }
            };

            function gameLoop() {
                if (isOver) return;
                setTimeout(() => {
                    ctx.fillStyle = '#121214';
                    ctx.fillRect(0, 0, 320, 320);

                    ctx.fillStyle = '#3b82f6';
                    ctx.fillRect(food.x, food.y, 14, 14);

                    let head = {x: snake[0].x + dx, y: snake[0].y + dy};
                    if (head.x < 0 || head.x >= 320 || head.y < 0 || head.y >= 320 || snake.some(s => s.x === head.x && s.y === head.y)) {
                        isOver = true;
                        playSfx('explosion');
                        ctx.fillStyle = '#ffffff';
                        ctx.font = 'bold 18px Inter';
                        ctx.fillText("Crash!", 130, 160);
                        return;
                    }
                    snake.unshift(head);

                    if (head.x === food.x && head.y === food.y) {
                        playSfx('success');
                        score++;
                        food = { x: Math.floor(Math.random() * 20) * 16, y: Math.floor(Math.random() * 20) * 16 };
                    } else {
                        snake.pop();
                    }

                    ctx.fillStyle = '#6366f1';
                    snake.forEach(s => ctx.fillRect(s.x, s.y, 14, 14));

                    gameLoop();
                }, 120);
            }
            gameLoop();
        }

                </div>
            `).join('');

        function filterApprovedMods() {
            const query = document.getElementById('mod-search-input').value.toLowerCase();
            const loader = document.getElementById('mod-loader-filter').value;
            const filtered = approvedMods.filter(m => {
                const matchesQuery = m.name.toLowerCase().includes(query) || m.desc.toLowerCase().includes(query);
                const matchesLoader = loader === 'all' || m.loader === loader;
                return matchesQuery && matchesLoader;
            });
            renderApprovedMods(filtered);
        }

            playSfx('click');
            const rand = approvedMods[Math.floor(Math.random() * approvedMods.length)];
            openModModal(rand.name, rand.desc);
        }

            playSfx('click');

            document.getElementById('modal-mod-desc').textContent = desc;
            document.getElementById('mod-modal').classList.remove('hidden');

        function renderMusicPlaylist(filtered = musicTracks) {
            const grid = document.getElementById('music-playlist-grid');

            `).join('');
        }

        function filterMusicGenre(genre) {
            playSfx('click');
            document.querySelectorAll('.music-genre-btn').forEach(el => {
                el.classList.remove('bg-indigo-600', 'text-white');
                el.classList.add('bg-[#2a2a32]', 'text-gray-300');
            });
            document.getElementById(`mgenre-${genre}`).classList.add('bg-indigo-600', 'text-white');
            document.getElementById(`mgenre-${genre}`).classList.remove('bg-[#2a2a32]', 'text-gray-300');

            const filtered = genre === 'all' ? musicTracks : musicTracks.filter(t => t.genre === genre);
            renderMusicPlaylist(filtered);
        }

        function playMusicTrack(index) {
            playSfx('click');
            currentAudioIndex = index;
            const track = musicTracks[index];
            document.getElementById('current-track-title').textContent = track.title;
            document.getElementById('current-track-artist').textContent = track.artist;
            const audio = document.getElementById('station-audio');
            audio.src = track.url;
            audio.play().catch(e => console.log("Audio autoplay prevented or stream loading:", e));
        }

        function playRandomMusicTrack() {
            const idx = Math.floor(Math.random() * musicTracks.length);
            playMusicTrack(idx);
        }

        function skipMusicTrack() {
            currentAudioIndex = (currentAudioIndex + 1) % musicTracks.length;
            playMusicTrack(currentAudioIndex);
        }
